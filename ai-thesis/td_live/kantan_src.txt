"""
Extension classes enhance TouchDesigner components with python. An
extension is accessed via ext.ExtensionClassName from any operator
within the extended component. If the extension is promoted via its
Promote Extension parameter, all its attributes with capitalized names
can be accessed externally, e.g. op('yourComp').PromotedFunction().

For more info, see: http://www.derivative.ca/wiki099/index.php?title=Extensions
"""

from TDStoreTools import StorageManager
TDF = op.TDModules.mod.TDFunctions
import json

class Projecter:
	"""
	Projecter description
	"""
	def __init__(self, ownerComp):
		# The component to which this extension is attached
		self.ownerComp = ownerComp

	def Collect(self):
		saveFile = ui.chooseFile(load=False, start=project.folder, fileTypes=['json'], title='Save Kantan Project')
		fullPathDat = op('project/fullPath')
		treeDat = op('project/tree')
		fullDict = {}
		fullDict['shapes'] = []
		fullDict['shapeDef'] = {}
		for i in treeDat.col('id'):
			if i.row > 0:
				shapeDict = {}
				sId = i.val
				shapeDict['sId'] = sId
				shapeDict['sType'] = treeDat[sId,'type'].val
				shapeDict['sOwner'] = treeDat[sId,'owner'].val
				shapeDict['sLayer'] = treeDat[sId,'layer'].val
				shapeDict['sLock'] = treeDat[sId,'lock'].val
				shapeDict['sVisible'] = treeDat[sId,'visible'].val
				shapeDict['sPath'] = fullPathDat[sId,'path'].val
				shape = op(shapeDict['sPath'])
				if shape:
					fullDict['shapes'].append(shapeDict)
					fullDict['shapeDef'][sId] = shape.GetDefinition()

		projJson = json.dumps(fullDict, sort_keys=True, indent=4, separators=(',', ': '))
		op('projectJson').text = projJson
		op('projectJson').save(saveFile)

	def Load(self):
		loadFile = ui.chooseFile(load=True, start=project.folder ,fileTypes=['json'], title='Load Kantan Project')
		projJson = open(loadFile,'r').read()
		projDef = json.loads(projJson)
		self.Clear()
		shapes = projDef.get('shapes',[])
		if shapes:
			treeDat = op('project/tree')
			fullPathDat = op('project/fullPath')
			for i in shapes:
				treeDat.appendRow([i['sId'],i['sType'],i['sOwner'],i['sLayer'],i['sLock'],i['sVisible']])
			shapes = sorted(shapes, key=lambda d: d['sOwner'])
			for i in shapes:
				op('project').ext.Layerizer.cookOwner(owner=int(i['sOwner']))
				shapePath = fullPathDat[i['sId'],'path'].val
				shapeOp = op(shapePath)
				if shapeOp:
					shapeOp.SetDefinition(defDict=projDef['shapeDef'][i['sId']], undo=False)
					shapeOp.cook(force=True)
				else:
					debug('something bad happend, Shape could not be loaded')
			self.ownerComp.par.Project = loadFile
		else:
			debug('there are no shapes to load')

	def Clear(self):
		treeDat = op('project/tree')
		treeDat.clear(keepFirstRow=True)
